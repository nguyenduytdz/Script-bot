-- 4. Logic Tìm & Chuyển Đến Server Ít Người Nhất
local isTeleporting = false

local function TeleportToSmallestServer()
    if isTeleporting then return end
    isTeleporting = true
    
    HopButton.Text = "ĐANG TÌM\nSERVER..."
    HopButton.BackgroundColor3 = Color3.fromRGB(80, 50, 20)

    local placeId = game.PlaceId
    local currentJobId = game.JobId
    local targetServerId = nil
    local cursor = ""

    -- Quét danh sách server công khai của game
    repeat
        local url = string.format("https://games.roblox.com/v1/games/%d/servers/Public?sortOrder=Asc&limit=100&cursor=%s", placeId, cursor)
        local success, result = pcall(function()
            return HttpService:JSONDecode(game:HttpGet(url))
        end)

        if success and result and result.data then
            -- Duyệt danh sách các server để tìm server có ít người nhất (1 người hoặc ít nhất có thể)
            for _, server in ipairs(result.data) do
                if server.playing < server.maxPlayers and server.id ~= currentJobId then
                    if server.playing <= 2 then -- Ưu tiên server có 1-2 người
                        targetServerId = server.id
                        break
                    end
                end
            end
            cursor = result.nextPageCursor or ""
        else
            break
        end

        task.wait(0.2)
    until targetServerId or cursor == ""

    -- Tiến hành chuyển server
    if targetServerId then
        HopButton.Text = "ĐANG CHUYỂN\nSERVER..."
        HopButton.BackgroundColor3 = Color3.fromRGB(20, 80, 30)
        TeleportService:TeleportToPlaceInstance(placeId, targetServerId, Players.LocalPlayer)
    else
        HopButton.Text = "KHÔNG THẤY\nTHỬ LẠI!"
        HopButton.BackgroundColor3 = Color3.fromRGB(100, 20, 20)
        task.wait(2)
        HopButton.Text = "JOIN SERVER\n1 NGƯỜI"
        HopButton.BackgroundColor3 = Color3.fromRGB(30, 24, 50)
        isTeleporting = false
    end
end

-- Lắng nghe sự kiện click nút
HopButton.MouseButton1Click:Connect(TeleportToSmallestServer)
